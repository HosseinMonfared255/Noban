    24→function hashStr(s: string): number {
    25→  let h = 0;
    26→  for (let i = 0; i < s.length; i++) {
    27→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
    28→  }
    29→  return h;
    30→}
    31→
    32→type SlotT = { time: string; booked: boolean };
    33→type DayT = {
    34→  name: string;
    35→  date: string;
    36→  status: "closed" | "full" | "open";
    37→  slots: SlotT[];
    38→};
    39→
    40→function startSaturday(): Date {
    41→  const t = new Date();
    42→  const d = t.getDay(); // 0 Sun .. 6 Sat
    43→  const add = (6 - d + 7) % 7;
    44→  const s = new Date(t);
    45→  s.setDate(t.getDate() + add);
    46→  return s;
    47→}
    48→const fmtDate = (d: Date) =>
    49→  new Intl.DateTimeFormat("fa-IR", { day: "numeric", month: "long" }).format(d);
    50→
    51→function buildSchedule(doctor: Doctor): DayT[] {
    52→  const start = startSaturday();
    53→  return DAY_NAMES.map((name, i) => {
    54→    const date = new Date(start);
    55→    date.setDate(start.getDate() + i);
    56→    const dateStr = fmtDate(date);
    57→    const isFriday = name === "جمعه";
    58→    const closed = isFriday || hashStr(doctor.name + "closed" + i) % 6 === 0;
    59→    if (closed) return { name, date: dateStr, status: "closed", slots: [] };
    60→    const fullDay = hashStr(doctor.name + "full" + i) % 4 === 1;
    61→    const slots = TIMES.map((time, k) => ({
    62→      time,
    63→      booked: fullDay || hashStr(doctor.name + i + time + k) % 3 === 0,
    64→    }));
    65→    const status: DayT["status"] = slots.every((s) => s.booked)
    66→      ? "full"
    67→      : "open";
    68→    return { name, date: dateStr, status, slots };
    69→  });
    70→}

   780→/* ---------------- live clinic status (patient view) ---------------- */
   781→type QItem = { id: number; name: string; status: "in" | "waiting" | "done"; you?: boolean };
   782→function LiveClinicModal({ doctor, onClose }: { doctor: Doctor; onClose: () => void }) {
   783→  const AVG = 7; // میانگین دقیقه‌ی هر ویزیت
   784→  const [queue, setQueue] = useState<QItem[]>([
   785→    { id: 1, name: "نگار حسینی", status: "in" },
   786→    { id: 2, name: "محمد رستمی", status: "waiting" },
   787→    { id: 3, name: "زهرا کاظمی", status: "waiting" },
   788→    { id: 4, name: "شما", status: "waiting", you: true },
   789→    { id: 5, name: "سینا ملکی", status: "waiting" },
   790→  ]);
   791→  const [secLeft, setSecLeft] = useState(AVG * 60);
   792→
   793→  useEffect(() => {
   794→    const t = setInterval(() => setSecLeft((s) => (s > 0 ? s - 1 : 0)), 1000);
   795→    return () => clearInterval(t);
   796→  }, []);
   797→  useEffect(() => {
   798→    if (secLeft === 0) {
   799→      setQueue((q) => {
   800→        const nq = q.map((x) => (x.status === "in" ? { ...x, status: "done" as const } : x));
   801→        const i = nq.findIndex((x) => x.status === "waiting");
   802→        if (i >= 0) nq[i] = { ...nq[i], status: "in" as const };
   803→        return nq;
   804→      });
   805→      setSecLeft(AVG * 60);
   806→    }
   807→  }, [secLeft]);
   808→
   809→  const current = queue.find((q) => q.status === "in");
   810→  const waiting = queue.filter((q) => q.status === "waiting");
   811→  const you = queue.find((q) => q.you);
   812→  const ahead = you ? waiting.indexOf(you) : 0;
   813→  const estMin = Math.max(1, ahead * AVG + Math.ceil(secLeft / 60));
   814→  const mm = String(Math.floor(secLeft / 60)).padStart(2, "0");
   815→  const ss = String(secLeft % 60).padStart(2, "0");
   816→
   817→  return (
   818→    <Modal onClose={onClose}>
   819→      <button onClick={onClose} data-cursor="hover" className="absolute left-4 top-4 z-10 grid h-9 w-9 place-items-center rounded-full bg-white/80 text-slate-600" aria-label="بستن"><Icon name="close" className="h-5 w-5" /></button>
   820→
   821→      {/* header */}
   822→      <div className="overflow-hidden rounded-2xl bg-gradient-to-l from-cyan-600 to-blue-700 p-4 text-white">
   823→        <div className="flex items-center justify-between">
   824→          <div><div className="text-[11px] text-cyan-50">{doctor.location}</div><div className="text-lg font-black">وضعیت زنده‌ی مطب</div></div>
   825→          <span className="flex items-center gap-1.5 rounded-lg bg-white/15 px-2.5 py-1.5 text-[11px] font-bold"><span className="h-2 w-2 animate-pulse rounded-full bg-emerald-300" />در حال پذیرش</span>
   826→        </div>
   827→        <div className="mt-2 flex items-center gap-4 text-sm">
   828→          <span>{toFa(waiting.length)} نفر در انتظار</span>
   829→          <span className="text-cyan-50">•</span>
   830→          <span>میانگین ویزیت {toFa(AVG)} دقیقه</span>
   831→        </div>
   832→      </div>
   833→
   834→      {/* current in room */}
   835→      <div className="mt-4 rounded-2xl border-2 border-cyan-500 bg-cyan-50 p-4">
   836→        <div className="mb-1 text-[11px] font-bold text-cyan-700">🔴 داخل اتاق دکتر</div>
   837→        {current ? (
   838→          <div className="flex items-center justify-between gap-2">
   839→            <div className="text-lg font-black text-slate-900">{current.name}</div>
   840→            <div className="flex items-center gap-2 rounded-xl bg-white px-3 py-1.5 text-sm font-black tabular-nums text-cyan-700"><Icon name="clock" className="h-4 w-4" />{toFa(mm)}:{toFa(ss)}</div>
   841→          </div>
   842→        ) : <div className="text-sm text-slate-500">در حال حاضر بیماری داخل نیست.</div>}
   843→      </div>

   848→          <div className="text-[11px] font-bold text-emerald-700">جایگاه شما در صف</div>
