   128→function exportCSV(filename: string, rows: (string | number)[][]) {
   129→  const bom = "\uFEFF";
   130→  const csv = rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(",")).join("\n");
   131→  const blob = new Blob([bom + csv], { type: "text/csv;charset=utf-8;" });
   132→  const url = URL.createObjectURL(blob);
   133→  const a = document.createElement("a");
   134→  a.href = url;
   135→  a.download = filename;
   136→  a.click();
   137→  URL.revokeObjectURL(url);
   138→}

   186→/* 3D bank card */
   187→function BankCard3D({ card, onEdit, onDelete }: { card: BankCard; onEdit: () => void; onDelete: () => void }) {
   188→  const ref = useRef<HTMLDivElement>(null);
   189→  const [t, setT] = useState({ rx: 0, ry: 0, gx: 50, gy: 50 });
   190→  const onMove = (e: React.MouseEvent) => {
   191→    const el = ref.current;
   192→    if (!el) return;
   193→    const r = el.getBoundingClientRect();
   194→    const px = (e.clientX - r.left) / r.width;
   195→    const py = (e.clientY - r.top) / r.height;
   196→    setT({ rx: -(py - 0.5) * 18, ry: (px - 0.5) * 22, gx: px * 100, gy: py * 100 });
   197→  };
   198→  const reset = () => setT({ rx: 0, ry: 0, gx: 50, gy: 50 });
   199→  const groups = card.number.replace(/\D/g, "").padEnd(16, "•").replace(/(.{4})/g, "$1 ").trim();
   200→  return (
   201→    <div style={{ perspective: 1100 }} className="group">
   202→      <div
   203→        ref={ref}
   204→        onMouseMove={onMove}
   205→        onMouseLeave={reset}
   206→        style={{ transform: `rotateX(${t.rx}deg) rotateY(${t.ry}deg)`, transformStyle: "preserve-3d" }}
   207→        className="relative aspect-[1.586] w-full rounded-2xl p-5 text-white shadow-2xl shadow-black/30 transition-transform duration-150"
   208→      >
   209→        <div className={`absolute inset-0 rounded-2xl bg-gradient-to-br ${gradFor(card.bank)}`} />
   210→        <div className="absolute inset-0 rounded-2xl opacity-70" style={{ background: `radial-gradient(circle at ${t.gx}% ${t.gy}%, rgba(255,255,255,.35), transparent 55%)` }} />
   211→        {/* actions */}
   212→        <div className="absolute left-3 top-3 z-10 flex gap-1.5 opacity-100 transition sm:opacity-0 sm:group-hover:opacity-100" style={{ transform: "translateZ(40px)" }}>
   213→          <button onClick={onEdit} data-cursor="hover" className="grid h-7 w-7 place-items-center rounded-lg bg-white/25 backdrop-blur hover:bg-white/40"><Icon name="edit" className="h-3.5 w-3.5" /></button>
   214→          <button onClick={onDelete} data-cursor="hover" className="grid h-7 w-7 place-items-center rounded-lg bg-white/25 backdrop-blur hover:bg-rose-500/70"><Icon name="trash" className="h-3.5 w-3.5" /></button>
   215→        </div>
   216→        {/* content */}
   217→        <div className="relative flex h-full flex-col justify-between" style={{ transform: "translateZ(30px)" }}>
   218→          <div className="flex items-start justify-between">
   219→            <div className="h-8 w-11 rounded-md bg-gradient-to-br from-amber-200 to-yellow-500 shadow-inner">
   220→              <div className="m-1 h-full rounded-sm border-t border-b border-amber-700/30" />
   221→            </div>
   222→            <span className="text-sm font-bold drop-shadow">{card.bank}</span>
   223→          </div>
   224→          <div className="text-base font-bold tabular-nums drop-shadow sm:text-lg" style={{ letterSpacing: "0.08em" }}>{toFa(groups)}</div>
   225→          <div className="flex items-end justify-between text-xs">
   226→            <div>
   227→              <div className="text-[9px] opacity-70">دارنده کارت</div>
   228→              <div className="font-bold">{card.holder}</div>
   229→            </div>
   230→            <div className="text-left">
   231→              <div className="text-[9px] opacity-70">انقضا</div>
   232→              <div className="font-bold tabular-nums">{toFa(card.expiry)}</div>
   233→            </div>
   234→          </div>
   235→        </div>
   236→      </div>
   237→    </div>
   238→  );
